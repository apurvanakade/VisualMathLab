About Visual Math Lab

How it works

Mathematics is full of ideas that come alive the moment you can see them move. Visual Math Lab makes them touchable: pick a function, drag a slider, change a starting guess, and watch the idea unfold one step at a time — from root finding and differential equations to probability, model fitting, and combinatorial proofs.

Explore on your own, use a page as a lecture demo, or hand one to students as a lab. Everything here is open source, so it is yours to fork, adapt, and extend.

// Matches the colors plotterMainPlot draws each trace in, so the overlay
// legend's swatches double as a color key.
// vmTheme is what makes the colors below reactive. The chart's own chrome
// (background, axes, gridlines) repaints on a dark-mode toggle by itself,
// via the relayout patch in plotly-fullscreen-button.js -- but the *trace*
// colors are values this page bakes into the traces it builds, so nothing
// repaints them unless mainPlot re-runs. Depending on vmTheme is what
// re-runs it. See VM.plotting.onThemeChange for why it fires a couple of
// frames after the toggle rather than inline.
vmTheme = Generators.observe(notify => VM.plotting.onThemeChange(notify))

chartColors = VM.plotting.colors(vmTheme)

plotterTraceColors = new Map([
  ["Curve", chartColors.fn],

// Applies the selected example's params to the fields above and clicks
// Plot, in place. A plain "change" listener attached once to the stable
// native <select> (rather than an OJS cell reactive on
// plotterExampleSelect's value) -- a <select> fires both "input" and
// "change" per selection, and a reactive cell would run twice, with the
// two overlapping calls racing each other.
applyPlotterExampleFromSelect = {
  const nativeSelect = (viewof plotterExampleSelect).querySelector("select")
  if (!nativeSelect) return
  nativeSelect.addEventListener("change", () => {
    const example = (viewof plotterExampleSelect).value
    if (!example) return
    VM.ui.applyExampleParams(plotterFieldSelectors, example.params, viewof plotterTrigger)
  })
}
// Plot commits the current formula/parameter-range fields -- edits above
// don't take effect until this is clicked.
viewof plotterTrigger = {
  const plotButton = Inputs.button("Plot", {value: 0, reduce: v => v + 1})
  plotButton.classList.add("ojs-auto", "ojs-end")
  return plotButton
}
// Pressing Enter in the formula/parameter-range fields clicks Plot instead
// of submitting the field's own (invisible) form.
enterToPlotterPlot = {
  for (const view of [viewof plotterXText, viewof plotterYText, viewof plotterTMin, viewof plotterTMax]) {
    const input = view.querySelector("input")
    if (!input) continue
    input.addEventListener("keydown", (e) => {
      if (e.key !== "Enter") return
      e.preventDefault()
      viewof plotterTrigger.querySelector("button").click()
    })
  }
}
// plotterCommitted snapshots the fields only when Plot is clicked -- it
// reads the stable `viewof` DOM views, not the reactive values, so it does
// not rerun on every keystroke.
plotterCommitted = {
  plotterTrigger
  const math = window.math
  return {
    xText: (viewof plotterXText).value,
    yText: (viewof plotterYText).value,
    tMin: VM.expressions.makeNumber(math, (viewof plotterTMin).value),
    tMax: VM.expressions.makeNumber(math, (viewof plotterTMax).value)
  }
}
// plotterSetup samples (x(t), y(t)) across the committed t-range. `key`
// changes whenever a new curve is committed, and is used as the plot's
// uirevision so a fresh curve autoscales instead of inheriting the
// previous curve's zoom.
plotterSetup = {
  const math = window.math
  const x = VM.expressions.makeFunctionOfT(math, plotterCommitted.xText)
  const y = VM.expressions.makeFunctionOfT(math, plotterCommitted.yText)

  let tMin = plotterCommitted.tMin
  let tMax = plotterCommitted.tMax
  const key = JSON.stringify(plotterCommitted)

  if (!x || !y || !Number.isFinite(tMin) || !Number.isFinite(tMax) || tMin === tMax) {
    return {ts: [], xs: [], ys: [], tMin: 0, tMax: 1, xRange: {lo: -1, hi: 1}, yRange: {lo: -1, hi: 1}, key}
  }
  if (tMin > tMax) { const temp = tMin; tMin = tMax; tMax = temp }

  // Parametric curves double back on themselves, so they need a good deal
  // more samples than a graph over an interval before the polyline stops
  // looking faceted.
  const sampleCount = 2400
  const ts = []
  const xs = []
  const ys = []
  for (let i = 0; i < sampleCount; i++) {
    const t = tMin + (tMax - tMin) * i / (sampleCount - 1)
    const xValue = x(t)
    const yValue = y(t)
    ts.push(t)
    if (Number.isFinite(xValue) && Number.isFinite(yValue) && Math.abs(xValue) < 1e8 && Math.abs(yValue) < 1e8) {
      xs.push(xValue)
      ys.push(yValue)
    } else {
      // A null in both coordinates breaks the trace here instead of
      // drawing a spurious segment across a pole or a gap in the domain.
      xs.push(null)
      ys.push(null)
    }
  }

  // Padded explicitly rather than left to Plotly's autorange: an axis pair
  // tied together by scaleanchor (see plotterMainPlot) autoranges with no
  // padding at all on the constraining axis, which put the square path
  // example's two vertical sides exactly on the plot's edges.
  const xRange = VM.plotting.paddedRange(xs, {relativePadding: 0.06, minPadding: 0.1})
  const yRange = VM.plotting.paddedRange(ys, {relativePadding: 0.06, minPadding: 0.1})
  return {ts, xs, ys, tMin, tMax, xRange, yRange, key}
}
// plotterMainPlot is a closure over a _plotDiv variable so the same DOM
// node is reused across reactive re-renders, preserving zoom/pan.
plotterMainPlot = {
  const Plotly = window.Plotly
  let _plotDiv = null

  return ({ts, xs, ys, xRange, yRange, key}, cursorIndex, showCurve, showCursor) => {
    const data = []

    if (showCurve) {
      data.push({
        x: xs, y: ys, type: "scatter", mode: "lines", name: "Curve", showlegend: false,
        line: {color: chartColors.fn, width: 2.5, shape: "spline"},
        customdata: ts,
        hovertemplate: "t = %{customdata:.3f}<br>x = %{x:.3f}<br>y = %{y:.3f}<extra></extra>"
      })
    }

    // The cursor reads straight out of the sampled arrays rather than
    // re-evaluating x(t)/y(t), so the marker always sits exactly on the
    // polyline that's drawn, never a hair off it.
    const i = cursorIndex
    if (showCursor && i !== null && i >= 0 && i < xs.length && xs[i] !== null) {
      data.push({
        x: [xs[i]], y: [ys[i]], type: "scatter", mode: "markers", name: "Point at t", showlegend: false,
        marker: {color: chartColors.alt, size: 12, line: {color: chartColors.halo, width: 2}},
        customdata: [ts[i]],
        hovertemplate: "t = %{customdata:.3f}<br>x = %{x:.3f}<br>y = %{y:.3f}<extra></extra>"
      })
    }
    const layout = {
      xaxis: {title: "x", zeroline: true, range: [xRange.lo, xRange.hi], constrain: "domain"},
      // scaleanchor keeps one unit of y the same length on screen as one
      // unit of x -- without it a circle plots as an ellipse, which ruins
      // every curve below.
      //
      // constrain: "domain" -- on both axes, or the unconstrained one still
      // gets widened -- makes Plotly satisfy that by shrinking the drawn
      // plot area to the curve's own aspect ratio, leaving the ranges above
      // exactly as asked for. The default ("range") instead widens
      // whichever range is too tight for the box -- and since newPlot runs
      // on a still-detached div (Plotly measures its 700x450 default) while
      // every later react runs at the real size, the two widened by
      // different amounts and the first slider nudge visibly jumped the
      // zoom. Pinning the ranges removes the discrepancy at the source.
      yaxis: {title: "y", zeroline: true, range: [yRange.lo, yRange.hi], scaleanchor: "x", scaleratio: 1, constrain: "domain"},
      hovermode: "closest",
      uirevision: key,
      autosize: true
    }
    const config = VM.plotting.config()

    if (!_plotDiv) {
      _plotDiv = document.createElement("div")
      _plotDiv.className = "plotly-box-large"
      Plotly.newPlot(_plotDiv, data, layout, config)
      VM.plotting.autoResize(_plotDiv)
    } else {
      Plotly.react(_plotDiv, data, layout, config)
    }
    return _plotDiv
  }
}
// The slider steps by one sample of plotterSetup, so its value doubles as
// an index into the sampled arrays -- see plotterMainPlot's cursor trace.
plotterCursorIndex = {
  if (plotterSetup.ts.length === 0) return null
  const span = plotterSetup.tMax - plotterSetup.tMin
  const index = Math.round((plotterT - plotterSetup.tMin) / span * (plotterSetup.ts.length - 1))
  return index
}
plotterShowCurve = plotterTraces.includes("Curve")
plotterShowCursor = plotterTraces.includes("Point at t")
plotterMainPlot(plotterSetup, plotterCursorIndex, plotterShowCurve, plotterShowCursor)
viewof plotterTraces = {
  const items = []
  for (const label of plotterTraceOptions) items.push({label, color: plotterTraceColors.get(label)})
  return VM.ui.legendOverlay(items, {value: plotterTraceOptions})
}
// Always-live: the cursor moves as the slider is dragged, no Plot click
// needed. Recreated whenever a new curve is committed, so its range always
// spans that curve's own t interval.
viewof plotterT = {
  const {tMin, tMax} = plotterSetup

  // Round the step to the nearest 1/2/5 x 10^k below span/500, so the
  // slider's own number readout stays short (0.05, not 0.0157145109...)
  // whatever t-range the current curve happens to span.
  const raw = (tMax - tMin) / 500
  const magnitude = Math.pow(10, Math.floor(Math.log10(raw)))
  const normalized = raw / magnitude
  let nice
  if (normalized <= 1) nice = 1
  else if (normalized <= 2) nice = 2
  else if (normalized <= 5) nice = 5
  else nice = 10
  const step = nice * magnitude

  const slider = Inputs.range([tMin, tMax], {label: "t", step, value: tMin})
  slider.classList.add("ojs-fill")
  return slider
}
