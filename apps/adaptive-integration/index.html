// integrand parses the committed formula and interval. It doesn't read the
// tolerance, so sweep, which reads it instead of result, doesn't rerun when
// the tolerance slider moves.
//   f — parsed function, or null for an unparseable formula
//   lo, hi — the interval, endpoints in increasing order (NaN when the
//     endpoints aren't finite and distinct)
integrand = {
  const f = VM.expressions.makeFunction(window.math, committed.fText)
  const a = committed.a0
  const b = committed.b0
  if (!f || !Number.isFinite(a) || !Number.isFinite(b) || a === b) return {f, lo: NaN, hi: NaN}
  return {f, lo: Math.min(a, b), hi: Math.max(a, b)}
}
// result: runs adaptive Simpson quadrature (method.js's adaptiveSimpson) on
// [a, b] at the current tolerance ε.
//   leaves — array of { a, b, mid, fa, fmid, fb, estimate, errorEstimate,
//     depth, width }, one entry per accepted subinterval, left to right
//   f — parsed function, reused for curve sampling in setup
//   lo, hi — the interval, endpoints in increasing order
//   totalEstimate — sum of leaf estimates (the adaptive quadrature's answer)
//   totalErrorEstimate — sum of leaf Richardson error estimates
//   trueIntegral — reference integral (method.js's referenceIntegral)
//   actualError — |totalEstimate - trueIntegral|
result = {
  const {f, lo, hi} = integrand
  const empty = {leaves: [], f: x => NaN, lo: NaN, hi: NaN, totalEstimate: NaN, totalErrorEstimate: NaN, trueIntegral: NaN, actualError: NaN}
  if (!f) return empty
  if (!Number.isFinite(lo)) return {...empty, f}

  const run = adaptiveSimpson(f, lo, hi, tolerance, maxDepth)
  if (!run) return {...empty, f}

  const trueIntegral = referenceIntegral(f, lo, hi)
  const actualError = Math.abs(run.totalEstimate - trueIntegral)
  return {leaves: run.leaves, f, lo, hi, totalEstimate: run.totalEstimate, totalErrorEstimate: run.totalErrorEstimate, trueIntegral, actualError}
}
// sweep reruns the adaptive rule at every tolerance on the slider's range
// (and the same rule on equal panels at equal cost) for the convergence
// plots. It reads integrand, not result, so it doesn't depend on the
// slider: moving it only moves the plots' current-ε markers.
sweep = {
  if (!Number.isFinite(integrand.lo)) return {rows: [], trueIntegral: NaN}
  return toleranceSweep(integrand.f, integrand.lo, integrand.hi, maxDepth)
}
// setup computes the data needed by the main chart: a dense quadratic
// (Simpson) sample per leaf for shading, the partition boundary points, and
// the axis ranges / smooth f(x) curve.
setup = {
  if (result.leaves.length === 0) {
    const empty = {lo: -3, hi: 3}
    return {panels: [], nodesX: [], nodesY: [], boundaries: [], maxDepthUsed: 0, sampleXs: [], sampleYs: [], xRange: empty, xBuffer: empty, yRange: {lo: -1, hi: 1}}
  }

  let maxDepthUsed = 0
  for (const leaf of result.leaves) {
    if (leaf.depth > maxDepthUsed) maxDepthUsed = leaf.depth
  }

  // Dense parabola samples per leaf, via the Lagrange quadratic through its
  // (a, fa), (mid, fmid), (b, fb) — the same curve Simpson's rule integrates
  // exactly on that leaf.
  const samplesPerLeaf = 16
  const panels = []
  for (const leaf of result.leaves) {
    const sample = VM.numerical.lagrangeQuadratic(leaf.a, leaf.fa, leaf.mid, leaf.fmid, leaf.b, leaf.fb, samplesPerLeaf)
    panels.push({xs: sample.xs, ys: sample.ys, depth: leaf.depth})
  }

  // Partition boundary points — every leaf's left edge, plus the final
  // right edge — used for the node markers and the vertical divider lines.
  const nodesX = [result.leaves[0].a]
  const nodesY = [result.leaves[0].fa]
  for (const leaf of result.leaves) {
    nodesX.push(leaf.b)
    nodesY.push(leaf.fb)
  }
  const boundaries = nodesX.slice(1, -1)

  const lo = result.leaves[0].a
  const hi = result.leaves[result.leaves.length - 1].b
  const xRange = VM.plotting.paddedRange([lo, hi], {emptyRange: [-3, 3], relativePadding: 0.1, minPadding: 0.25})

  // A wider domain (double the padding of xRange) so the curve is already
  // sampled beyond the initial view — panning/zooming out reveals more of
  // the curve instead of blank canvas. The initial view stays xRange.
  const xBuffer = VM.plotting.paddedRange([lo, hi], {emptyRange: [-6, 6], relativePadding: 0.25, minPadding: 0.5})

  // Sample f across the buffer range for the smooth curve trace.
  const sampleCount = 700
  const sampleXs = Array.from({length: sampleCount}, (_, i) => xBuffer.lo + (xBuffer.hi - xBuffer.lo) * i / (sampleCount - 1))
  const sampleYs = []
  for (const x of sampleXs) {
    const y = result.f(x)
    if (Number.isFinite(y) && Math.abs(y) < 1e8) {
      sampleYs.push(y)
    } else {
      sampleYs.push(null)
    }
  }

  // y-range from f over [lo, hi] (plus 0, where the panels stand) only, not
  // the tolerance-dependent nodes and panels, so moving the tolerance slider
  // never rescales the axis.
  const yValues = [0]
  for (let i = 0; i < sampleXs.length; i++) {
    const x = sampleXs[i]
    if (x >= xRange.lo && x <= xRange.hi && sampleYs[i] !== null) yValues.push(sampleYs[i])
  }
  const yRange = VM.plotting.paddedRange(yValues, {emptyRange: [-1, 1], relativePadding: 0.25, minPadding: 0.5})

  return {panels, nodesX, nodesY, boundaries, maxDepthUsed, sampleXs, sampleYs, xRange, xBuffer, yRange}
}

      }
    }

    const layout = {
      xaxis: {title: "x", range: [data.xRange.lo, data.xRange.hi], zeroline: true},
      yaxis: {title: "y", range: [data.yRange.lo, data.yRange.hi], zeroline: true},
      shapes,
      hovermode: "closest",
      uirevision: "static",  // keeps zoom/pan state across reactive updates
      autosize: true
    }
    const config = VM.plotting.config()

    // First render: create the div. Later renders: update in place (preserves zoom).

      VM.plotting.autoResize(_plotDiv)
    } else {
      Plotly.react(_plotDiv, traces, layout, config)
    }
    return _plotDiv
  }
}

How it works

Numerical integration lays a fixed grid over \([a,b]\) and uses the same panel width everywhere. An adaptive method instead subdivides only where it has to: narrow panels where \(f\) wiggles or turns sharply, wide ones where \(f\) is nearly a parabola already.

To decide, it needs an estimate of the error a panel makes. Simpson’s rule on \([p,q]\) with midpoint \(m = (p+q)/2\) is

\[ S[p,q] = \frac{q-p}{6}\Bigl(f(p) + 4f(m) + f(q)\Bigr), \]

and the same panel split in half gives a second, finer estimate \(S[p,m] + S[m,q]\). Since Simpson’s error scales like \(h^4\), halving \(h\) shrinks it by a factor of \(16\), so the difference between the two is almost all error:

\[ E = \frac{\bigl|\, S[p,m] + S[m,q] - S[p,q] \,\bigr|}{15}. \]

It is then compared with the tolerance \(\varepsilon\) set by the slider:

  • Accept if \(E \le \varepsilon\): keep \([p,q]\) as a panel of the final partition, using the finer estimate \(S[p,m] + S[m,q]\).
  • Refine if \(E > \varepsilon\): bisect at \(m\) and recurse on each half with \(\varepsilon/2\), so the total error stays bounded by the original tolerance.

A larger \(\varepsilon\) means fewer, wider panels and a coarser partition; a smaller \(\varepsilon\) means more panels, clustered exactly where \(f\) is hardest to approximate.

NoteConvergence plots 
// toleranceErrorPlot draws the actual and the estimated error as multiples
// of the tolerance they were run at, log₁₀ (error / ε) against the exponent
// e. The line at 0 is error = ε and the band above it is where ε was
// missed. Plotting the errors themselves would show three parallel lines:
// the stopping rule ties both to ε. A labelled vertical rule marks the
// slider's ε;
// a grey band, named along its edge, the tolerances at which some panel
// stopped at Max depth.
toleranceErrorPlot = {
  const actualName = "Actual error / ε"
  const estimatedName = "Estimated error Σ E / ε"
  const actual = tolerancePoints(sweep.rows, sweep.trueIntegral, "adaptiveError")
  const estimated = tolerancePoints(sweep.rows, sweep.trueIntegral, "estimatedError")
  const firstExponent = toleranceExponents[0]
  const lastExponent = toleranceExponents[toleranceExponents.length - 1]

  // The y range always spans [-2, 1], so the band above 0 never fills the
  // chart, and grows to hold every point.
  let top = 1
  let bottom = -2
  for (const point of actual.concat(estimated)) {
    top = Math.max(top, point.y + 0.3)
    bottom = Math.min(bottom, point.y - 0.3)
  }

  const marks = [
    Plot.rect([{x1: firstExponent, x2: lastExponent, y1: 0, y2: top}], {x1: "x1", x2: "x2", y1: "y1", y2: "y2", fill: chartColors.alt, fillOpacity: 0.12}),
    Plot.text([{x: firstExponent, y: top}], {x: "x", y: "y", text: () => "error > ε", fill: chartColors.alt, textAnchor: "start", lineAnchor: "top", dx: 4, dy: 4}),
    Plot.ruleY([0], {stroke: chartColors.ink, strokeDasharray: "4,3"}),
    Plot.ruleX([tolExponent], {stroke: chartColors.muted}),
    Plot.text([tolExponent], {x: d => d, frameAnchor: "top", text: () => "current ε", fill: chartColors.muted, textAnchor: tolExponent > 8 ? "end" : "start", dx: tolExponent > 8 ? -4 : 4, dy: 16}),
    Plot.line(estimated, {x: "exponent", y: "y", stroke: () => estimatedName}),
    Plot.dot(estimated, {x: "exponent", y: "y", fill: () => estimatedName, r: 3}),
    Plot.line(actual, {x: "exponent", y: "y", stroke: () => actualName}),
    Plot.dot(actual, {x: "exponent", y: "y", fill: () => actualName, r: 3})
  ]

  // A panel accepted only because it reached Max depth may miss its share
  // of ε. Mark the first ε at which that happens.
  let firstLimited = null
  for (const row of sweep.rows) {
    if (row.depthLimited) {
      firstLimited = row
      break
    }
  }
  if (firstLimited) {
    // A shaded band over every ε from there on, named along its left edge.
    const labelInside = firstLimited.exponent <= lastExponent - 0.5
    marks.push(
      Plot.rect([{x1: firstLimited.exponent, x2: lastExponent, y1: bottom, y2: top}], {x1: "x1", x2: "x2", y1: "y1", y2: "y2", fill: chartColors.muted, fillOpacity: 0.18}),
      Plot.ruleX([firstLimited.exponent], {stroke: chartColors.muted}),
      Plot.text([firstLimited], {x: "exponent", y: () => bottom, text: () => "stopped at Max depth", fill: chartColors.ink, rotate: -90, textAnchor: "start", lineAnchor: labelInside ? "top" : "bottom", dx: labelInside ? 4 : -4, dy: -6})
    )
  }

  const colorDomain = [actualName, estimatedName]
  const colorRange = [chartColors.ok, chartColors.warn]
  const chart = Plot.plot({
    width: 340, height: 300, marginLeft: 46,
    x: {label: "e (ε = 10⁻ᵉ)", domain: [firstExponent, lastExponent]},
    y: {label: "log₁₀ (error / ε)", domain: [bottom, top]},
    color: {domain: colorDomain, range: colorRange},
    marks
  })
  const container = document.createElement("div")
  container.append(chart, Plot.legend({color: {domain: colorDomain, range: colorRange}}))
  return container
}
// costErrorPlot draws log₁₀ error against log₁₀ (evaluations of f) for the
// adaptive rule and the same rule on a uniform grid at equal cost, one point
// per tolerance, with a labelled vertical rule at the slider's ε, as in
// toleranceErrorPlot (both curves have a point there, at the same cost).
costErrorPlot = {
  const adaptiveName = "Adaptive panels"
  const uniformName = "Uniform panels, same cost"
  const adaptive = logPoints(sweep.rows, sweep.trueIntegral, "evaluations", "adaptiveError")
  const uniform = logPoints(sweep.rows, sweep.trueIntegral, "evaluations", "uniformError")
  // Both curves have a point at the slider's ε, at the same cost.
  const current = []
  for (const point of adaptive) {
    if (point.exponent === tolExponent) current.push(point.x)
  }
  const lastCost = adaptive.length ? adaptive[adaptive.length - 1].x : 0
  const labelLeft = current.length > 0 && current[0] > 0.75 * lastCost

  const marks = [
    Plot.ruleX(current, {stroke: chartColors.muted}),
    Plot.text(current, {x: d => d, frameAnchor: "top", text: () => "current ε", fill: chartColors.muted, textAnchor: labelLeft ? "end" : "start", dx: labelLeft ? -4 : 4, dy: 4}),
    Plot.line(uniform, {x: "x", y: "y", stroke: () => uniformName}),
    Plot.dot(uniform, {x: "x", y: "y", fill: () => uniformName, r: 3}),
    Plot.line(adaptive, {x: "x", y: "y", stroke: () => adaptiveName}),
    Plot.dot(adaptive, {x: "x", y: "y", fill: () => adaptiveName, r: 3})
  ]

  const colorDomain = [adaptiveName, uniformName]
  const colorRange = [chartColors.ok, chartColors.alt]
  const chart = Plot.plot({
    width: 340, height: 300, marginLeft: 46,
    x: {label: "log₁₀ evaluations of f"},
    y: {label: "log₁₀ error"},
    color: {domain: colorDomain, range: colorRange},
    marks
  })
  const container = document.createElement("div")
  container.append(chart, Plot.legend({color: {domain: colorDomain, range: colorRange}}))
  return container
}
