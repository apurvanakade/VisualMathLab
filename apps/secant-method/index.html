Secant Method

Approximates Newton’s method without a derivative, using the secant line through the last two iterates, or inverse quadratic interpolation through the last three.

// inverseInterpolant evaluates, at height y, the polynomial x = q(y) that
// passes through every node (x_k, f(x_k)) -- Lagrange interpolation with the
// roles of x and y swapped. Through two nodes q is the secant line; through
// three it is the sideways parabola of inverse quadratic interpolation.
// Either way the next iterate is q(0), where the interpolant meets the x-axis.
function inverseInterpolant(nodes, y) {
  let x = 0
  for (let j = 0; j < nodes.length; j++) {
    let term = nodes[j].x
    for (let k = 0; k < nodes.length; k++) {
      if (k === j) continue
      term *= (y - nodes[k].fx) / (nodes[j].fx - nodes[k].fx)
    }
    x += term
  }
  return x
}
// runMethod iterates from the given initial guesses: two for the secant
// method, three for inverse quadratic interpolation (each step interpolates
// the last guesses.length points). Returns
//   rows   — array of { i, nodes, xNext, error }, one entry per iteration;
//            nodes are the points interpolated at that step, oldest first,
//            each { i, x, fx }
//   points — every x visited so far, x_0, x_1, ..., x_N (used for the
//            iterates plot, which needs the initial guesses too)
function runMethod(f, initialGuesses, maxIterations) {
  for (const guess of initialGuesses) {
    if (!Number.isFinite(guess)) return {rows: [], points: []}
  }

  const nodeCount = initialGuesses.length
  const points = []
  for (const x of initialGuesses) points.push({x, fx: f(x)})
  const rows = []

  let i = nodeCount - 1
  while (rows.length < maxIterations) {
    const nodes = []
    for (let k = i - nodeCount + 1; k <= i; k++) nodes.push({i: k, x: points[k].x, fx: points[k].fx})
    const curr = nodes[nodes.length - 1]

    // Stop if any node's f-value blew up — the method has diverged.
    let diverged = false
    for (const node of nodes) {
      if (!Number.isFinite(node.fx)) diverged = true
    }
    if (diverged) {
      rows.push({i, nodes, xNext: NaN, error: NaN})
      return {rows, points}
    }

    // Stop if a node is already an exact root.
    let root = null
    for (const node of nodes) {
      if (node.fx === 0 && root === null) root = node
    }
    if (root) {
      rows.push({i, nodes, xNext: root.x, error: 0})
      return {rows, points}
    }

    // Stop if two nodes share (nearly) the same f-value — the interpolant
    // x = q(y) would have to divide by their difference.
    let degenerate = false
    for (let a = 0; a < nodes.length; a++) {
      for (let b = a + 1; b < nodes.length; b++) {
        if (Math.abs(nodes[a].fx - nodes[b].fx) < 1e-14) degenerate = true
      }
    }
    if (degenerate) {
      rows.push({i, nodes, xNext: NaN, error: NaN})
      return {rows, points}
    }

    const xNext = inverseInterpolant(nodes, 0)
    const error = Math.abs(xNext - curr.x)
    rows.push({i, nodes, xNext, error})

    points.push({x: xNext, fx: f(xNext)})
    i++
  }
  return {rows, points}
}
// result: runs the selected method with the current committed inputs.
//   rows, points — see runMethod
//   f      — parsed function, reused for curve sampling in setup
//   method — "secant" or "iqi", the method these rows came from
result = {
  const math = window.math

  // Parse f(x) from the committed string. makeFunction returns null if the
  // expression is invalid.
  const f = VM.expressions.makeFunction(math, committed.fText)
  const fallback = x => NaN

  if (!f) return {rows: [], points: [], f: fallback, method}

  // Secant interpolates the last two points, IQI the last three.
  let initialGuesses = [committed.x0, committed.x1]
  if (method === "iqi") initialGuesses = [committed.x0, committed.x1, committed.x2]

  const run = runMethod(f, initialGuesses, maxIterations)
  return {rows: run.rows, points: run.points, f, method}
}
// comparison runs both methods on the committed inputs, whatever the Method
// toggle says, so the convergence plots can show them side by side.
comparison = {

  const runs = [
    {label: "Secant", color: chartColors.fn, guesses: [committed.x0, committed.x1]},
    {label: "Inverse quadratic", color: chartColors.alt, guesses: [committed.x0, committed.x1, committed.x2]}
  ]
  for (const run of runs) {
    let outcome = {rows: [], points: []}
    if (f) outcome = runMethod(f, run.guesses, maxIterations)
    run.rows = outcome.rows
    run.points = outcome.points
  }
  return runs
}
// setup computes the data shared by all three plots below: which rows are
// visible at the current step, and the axis ranges / curve samples for the
// main chart.
setup = {
  const visibleRows = result.rows.slice(0, Math.min(Number(stepControl), result.rows.length) + 1)

  // Axis ranges are computed from ALL rows, not just visibleRows,
  // so the axes stay fixed as the user steps through iterations.
  const xValues = []
  for (const point of result.points) xValues.push(point.x)
  for (const row of result.rows) xValues.push(row.xNext)
  const xRange = VM.plotting.paddedRange(xValues, {emptyRange: [-3, 3], relativePadding: 0.25, minPadding: 0.5})

  // A wider domain (double the padding of xRange) so the curve is already
  // sampled beyond the initial view — panning/zooming out reveals more of
  // the curve instead of blank canvas. The initial view stays xRange.
  const xBuffer = VM.plotting.paddedRange(xValues, {emptyRange: [-6, 6], relativePadding: 0.5, minPadding: 1.0})

  // Sample f across the buffer range for the smooth curve trace.
  const sampleCount = 700
  const sampleXs = Array.from({length: sampleCount}, (_, i) => xBuffer.lo + (xBuffer.hi - xBuffer.lo) * i / (sampleCount - 1))
  const sampleYs = []
  for (const x of sampleXs) {
    const y = result.f(x)
    if (Number.isFinite(y) && Math.abs(y) < 1e8) {
      sampleYs.push(y)
    } else {
      sampleYs.push(null)
    }
  }

  // Collect y values for the y-axis range, including every interpolation
  // node. Only samples within the initial view (xRange) count, so yRange —
  // and the initial view — stay unaffected by the wider sampling.
  // Cap at 1e8 so a runaway iterate doesn't blow up the axis range.
  const yValues = [0]
  for (let i = 0; i < sampleXs.length; i++) {
    const x = sampleXs[i]
    if (x >= xRange.lo && x <= xRange.hi && sampleYs[i] !== null) yValues.push(sampleYs[i])
  }
  for (const row of result.rows) {
    for (const node of row.nodes) {
      if (Number.isFinite(node.fx) && Math.abs(node.fx) <= 1e8) yValues.push(node.fx)
    }
  }
  const yRange = VM.plotting.paddedRange(yValues, {emptyRange: [-1, 1], relativePadding: 0.25, minPadding: 0.5})

  return {visibleRows, xRange, xBuffer, yRange, sampleXs, sampleYs, method: result.method}
}
// interpolantCurve samples x = q(y) (see inverseInterpolant) for one row,
// over the f-values of its nodes and 0, plus a margin on either side, so the
// drawn curve always runs through every node and the x-intercept. Returns
// null when the row has no usable interpolant (diverged or degenerate).
function interpolantCurve(row) {
  if (!Number.isFinite(row.xNext)) return null
  let yLo = 0
  let yHi = 0
  for (const node of row.nodes) {
    if (!Number.isFinite(node.x) || !Number.isFinite(node.fx)) return null
    yLo = Math.min(yLo, node.fx)
    yHi = Math.max(yHi, node.fx)
  }
  const margin = 0.25 * (yHi - yLo)
  yLo -= margin
  yHi += margin

  const sampleCount = 200
  const xs = []
  const ys = []
  for (let k = 0; k < sampleCount; k++) {
    const y = yLo + (yHi - yLo) * k / (sampleCount - 1)
    const x = inverseInterpolant(row.nodes, y)
    if (Number.isFinite(x)) {
      xs.push(x)
      ys.push(y)
    }
  }
  return {xs, ys}
}
// mainPlot draws f(x), the interpolant (secant line or inverse quadratic),
// the interpolation points, and the x-intercepts.
//
// _plotDiv is kept in a closure so Plotly reuses the same DOM node across
// reactive updates — this preserves zoom and pan when the step changes.
mainPlot = {
  const Plotly = window.Plotly
  let _plotDiv = null

  return ({visibleRows, xRange, xBuffer, yRange, sampleXs, sampleYs, method}, showFCurve, showSecant, showSecantPoints, showIntercepts) => {
    const data = []

    // Hover name only — the legend label stays "Interpolant" for both methods.
    let interpolantName = "Secant line"
    if (method === "iqi") interpolantName = "Inverse quadratic"

    if (showFCurve) {
      // The function curve
      data.push({x: sampleXs, y: sampleYs, type: "scatter", mode: "lines", name: "f(x)", showlegend: false, line: {color: chartColors.fn, width: 3}})
    }

    // Every point used so far: all nodes of the first visible row, then
    // the newest node of each later row.
    const secantPoints = []
    for (let r = 0; r < visibleRows.length; r++) {
      const nodes = visibleRows[r].nodes
      let first = nodes.length - 1
      if (r === 0) first = 0
      for (let k = first; k < nodes.length; k++) {
        if (Number.isFinite(nodes[k].x) && Number.isFinite(nodes[k].fx)) secantPoints.push(nodes[k])
      }
    }

    // Dotted vertical lines from the x-axis up to each interpolation point
    if (showSecantPoints) {
      for (const p of secantPoints) {
        data.push({x: [p.x, p.x], y: [0, p.fx], type: "scatter", mode: "lines", showlegend: false, hoverinfo: "skip", line: {color: chartColors.muted, width: 1.5, dash: "dot"}})
      }
    }

    // Interpolants: previous ones faded, current one solid. Each spans its
    // nodes and the x-intercept (plus a margin), so the intercept marker is
    // always on the drawn curve.
    const curves = []
    for (const r of visibleRows) {
      const curve = interpolantCurve(r)
      if (curve) curves.push(curve)
    }

    if (showSecant) {
      for (let k = 0; k < curves.length - 1; k++) {
        data.push({
          x: curves[k].xs, y: curves[k].ys,
          type: "scatter", mode: "lines", showlegend: false, hoverinfo: "skip",
          line: {color: "rgba(220, 38, 38, 0.18)", width: 1.5}
        })
      }
      if (curves.length > 0) {
        const current = curves[curves.length - 1]
        data.push({
          x: current.xs, y: current.ys,
          type: "scatter", mode: "lines", name: interpolantName, showlegend: false,
          line: {color: chartColors.alt, width: 2.5}
        })
      }
    }

    // Red markers at each (x_i, f(x_i)) interpolation point
    if (showSecantPoints && secantPoints.length > 0) {
      const xs = [], ys = [], labels = []
      for (const p of secantPoints) {
        xs.push(p.x)
        ys.push(p.fx)
        labels.push(`x${VM.plotting.subscript(p.i)}`)
      }
      data.push({
        x: xs, y: ys,
        type: "scatter", mode: "markers+text", name: "Interpolation points",
        text: labels, textposition: "top center", showlegend: false,
        marker: {color: chartColors.alt, size: 10, symbol: "circle", line: {color: chartColors.halo, width: 1}}
      })
    }

    // Green markers at each x-intercept (x_{i+1})
    const intercepts = []
    for (const r of visibleRows) {
      if (Number.isFinite(r.xNext)) intercepts.push({i: r.i + 1, x: r.xNext})
    }
    if (showIntercepts && intercepts.length > 0) {
      const xs = [], ys = [], labels = []
      for (const r of intercepts) {
        xs.push(r.x)
        ys.push(0)
        labels.push(`x${VM.plotting.subscript(r.i)}`)
      }
      data.push({
        x: xs, y: ys,
        type: "scatter", mode: "markers+text", name: "x-intercepts",
        text: labels, textposition: "bottom center", showlegend: false,
        marker: {color: chartColors.ok, size: 10, symbol: "circle", line: {color: chartColors.halo, width: 1}}
      })
    }

    const layout = {
      xaxis: {title: "x", range: [xRange.lo, xRange.hi], zeroline: true},
      yaxis: {title: "f(x)", range: [yRange.lo, yRange.hi], zeroline: true},
      hovermode: "closest",
      uirevision: "static",  // keeps zoom/pan state across reactive updates
      autosize: true
    }
    const config = VM.plotting.config()

    // First render: create the div. Later renders: update in place (preserves zoom).
    if (!_plotDiv) {
      _plotDiv = document.createElement("div")
      _plotDiv.className = "plotly-box-large"
      Plotly.newPlot(_plotDiv, data, layout, config)
      // newPlot fires before the div is in the DOM, so Plotly measures 0 width.
      // A ResizeObserver catches the real size once it's inserted and laid out,
      // and keeps correcting it on any later layout/viewport change.
      VM.plotting.autoResize(_plotDiv)
    } else {
      Plotly.react(_plotDiv, data, layout, config)
    }
    return _plotDiv
  }
}

// iteratesPlot draws xₙ vs n for the entire run of both methods, including
// their initial guesses — it doesn't depend on the current step, so it's
// just plotted once per run.
iteratesPlot = {
  const pointRows = []
  for (const run of comparison) {
    for (let k = 0; k < run.points.length; k++) {
      const point = run.points[k]
      if (Number.isFinite(point.x)) pointRows.push({method: run.label, i: k, x: point.x})
    }
  }

  const domain = [], range = []
  for (const run of comparison) {
    domain.push(run.label)
    range.push(run.color)
  }

// convergencePlot draws log₁₀|eₙ₊₁| vs log₁₀|eₙ| (successive step sizes,
// eₙ = |xₙ₊₁ − xₙ|) for both methods on one chart, each with its own linear
// regression line, for the entire run. Both methods converge superlinearly
// for a simple root: the secant method with order φ = (1+√5)/2 ≈ 1.618 (the
// golden ratio), inverse quadratic interpolation with order ≈ 1.839 — both
// slower than Newton's quadratic order but without needing a derivative.
// This log-log successive-error plot is the standard diagnostic: its slope
// estimates the order of convergence p directly.
convergencePlot = {
  const marks = []
  const colorDomain = [], colorRange = []
  // legendDomain/legendRange name each method with its fitted order; the
  // dots share their method's color, so they need no legend entry of their own.
  const legendDomain = [], legendRange = []

  for (const run of comparison) {
    const logRows = []
    for (let k = 0; k < run.rows.length - 1; k++) {
      const en  = run.rows[k].error
      const en1 = run.rows[k + 1].error
      if (Number.isFinite(en) && en > 0 && Number.isFinite(en1) && en1 > 0) {
        logRows.push({x: Math.log10(en), y: Math.log10(en1)})
      }
    }

    let name = run.label
    const fit = VM.numerical.linearRegression(logRows)
    if (fit) {
      name = `${run.label}: order ≈ ${fit.slope.toFixed(2)}`
      marks.push(Plot.line(
        [{x: fit.xlo, y: fit.slope * fit.xlo + fit.intercept}, {x: fit.xhi, y: fit.slope * fit.xhi + fit.intercept}],
        {x: "x", y: "y", stroke: () => name, strokeDasharray: "6,4"}
      ))
    }
    marks.push(Plot.dot(logRows, {x: "x", y: "y", fill: () => name, r: 4}))
    colorDomain.push(name)
    colorRange.push(run.color)
    legendDomain.push(name)
    legendRange.push(run.color)
  }

  const chart = Plot.plot({
    width: 340, height: 300, marginLeft: 46,
    x: {label: "log₁₀|eₙ|"},
    y: {label: "log₁₀|eₙ₊₁|"},
    color: {domain: colorDomain, range: colorRange},
    marks
  })

  const container = document.createElement("div")
  container.append(chart)
  container.append(Plot.legend({color: {domain: legendDomain, range: legendRange}}))
  return container
}

Both methods find a root of \(f(x)=0\) without a derivative, by interpolating the most recent iterates and taking the point where the interpolant crosses the \(x\)-axis as the next iterate. Write \(y_k = f(x_k)\).

The secant method uses the line through the last two points \((x_{n-1}, y_{n-1})\) and \((x_n, y_n)\), replacing Newton’s tangent line with a secant:

\[ x_{n+1}=x_n-y_n\,\frac{x_n-x_{n-1}}{y_n-y_{n-1}}. \]

Inverse quadratic interpolation (IQI) uses the last three points, and fits \(x\) as a quadratic function of \(y\), a sideways parabola \(x=q(y)\), rather than \(y\) as a function of \(x\). The next iterate is \(x_{n+1}=q(0)\):

\[ x_{n+1}=\frac{x_{n-2}\,y_{n-1}\,y_n}{(y_{n-2}-y_{n-1})(y_{n-2}-y_n)}+\frac{x_{n-1}\,y_{n-2}\,y_n}{(y_{n-1}-y_{n-2})(y_{n-1}-y_n)}+\frac{x_n\,y_{n-2}\,y_{n-1}}{(y_n-y_{n-2})(y_n-y_{n-1})}. \]

Because a sideways parabola crosses the \(x\)-axis exactly once, there is no need to solve a quadratic equation. (Read the same way, the secant line is \(x\) as a linear function of \(y\), evaluated at \(y=0\).)

Near a simple root the secant method converges with order \(\varphi=(1+\sqrt5)/2\approx 1.62\) and IQI with order about \(1.84\), both between bisection’s linear rate and Newton’s quadratic one. The slope of the log-log plot in “Convergence plots” estimates this order. Each method breaks down when two of the interpolated \(f\)-values coincide.
