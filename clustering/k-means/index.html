// basePoints: trueGroups Gaussian blobs of pointsPerGroup points each. Blob
// centers and each point's unit-normal offset are drawn once per (seed,
// regenerate, trueGroups) combination — the spread slider below only
// scales these fixed offsets, so it stays a smooth live control instead of
// redrawing new randomness on every tick (same trick as the noise slider on
// the Polynomial Fits page).
basePoints = {
  regenerateTrigger
  const rand = VM.sampling.seededRandom((Math.floor(seedValue) + regenerateTrigger * 1000003 + trueGroups * 97) >>> 0)
  const gaussian = VM.sampling.gaussianRandom(rand)

  const centers = []
  for (let g = 0; g < trueGroups; g++) {
    centers.push({x: -8 + 16 * rand(), y: -8 + 16 * rand()})
  }

  const points = []
  for (let g = 0; g < trueGroups; g++) {
    for (let i = 0; i < pointsPerGroup; i++) {
      points.push({group: g, cx: centers[g].x, cy: centers[g].y, ox: gaussian(), oy: gaussian()})
    }
  }
  return {centers, points}
}
// points: each point sits at its blob center plus its fixed unit-normal
// offset scaled by the spread slider — all coincident with their center at
// spread 0, drifting further apart as spread increases.
points = {
  const pts = []
  for (const p of basePoints.points) {
    pts.push({x: p.cx + spread * p.ox, y: p.cy + spread * p.oy, trueGroup: p.group})
  }
  return pts
}
// pointOrder: a fixed random arrival order for Online mode, reshuffled
// whenever the seed changes or new points are drawn — but NOT when k
// changes, since which point arrives when shouldn't depend on how many
// clusters you're asking for.
pointOrder = {
  const rand = VM.sampling.seededRandom((Math.floor(seedValue) + regenerateTrigger * 1000003 + 424243) >>> 0)
  const order = []
  for (let i = 0; i < points.length; i++) order.push(i)
  for (let i = order.length - 1; i > 0; i--) {
    const j = Math.floor(rand() * (i + 1))
    const tmp = order[i]
    order[i] = order[j]
    order[j] = tmp
  }
  return order
}
distSq = (a, b) => {
  const dx = a.x - b.x, dy = a.y - b.y
  return dx * dx + dy * dy
}
// runBatchKMeans: Lloyd's algorithm.
//   rows — array of { iter, centroids, assignments, inertia, reassigned },
//   one entry per iteration. `centroids`/`assignments` are the state used
//   to produce that row (i.e. the centroids the assignment step read from),
//   matching how Newton's method rows record x_n before the update to
//   x_{n+1} — so the step slider shows "here's what iteration n saw."
//   Stops early once no point's assignment changes from the previous
//   iteration (converged), or at maxIterations.
function runBatchKMeans(points, k, maxIterations, seedValue, regenerateTrigger) {
  // Initial centroids: k distinct data points, chosen with a seeded RNG of
  // its own (not a continuation of basePoints' RNG) so changing k always
  // starts from a reproducible draw rather than just extending whatever
  // sequence point generation happened to leave off at.
  const initRand = VM.sampling.seededRandom((Math.floor(seedValue) + regenerateTrigger * 1000003 + k * 7919) >>> 0)
  const order = []
  for (let i = 0; i < points.length; i++) order.push(i)
  for (let i = 0; i < Math.min(k, order.length); i++) {
    const j = i + Math.floor(initRand() * (order.length - i))
    const tmp = order[i]
    order[i] = order[j]
    order[j] = tmp
  }
  let centroids = []
  for (let c = 0; c < k; c++) {
    const idx = order[c % order.length]
    centroids.push({x: points[idx].x, y: points[idx].y})
  }

  const rows = []
  let prevAssignments = null
  let iter = 0
  while (iter < maxIterations) {
    // Assignment step: each point joins its nearest centroid.
    const assignments = []
    for (const p of points) {
      let best = 0, bestDist = Infinity
      for (let c = 0; c < centroids.length; c++) {
        const d = distSq(p, centroids[c])
        if (d < bestDist) {
          bestDist = d
          best = c
        }
      }
      assignments.push(best)
    }

    let inertia = 0
    for (let i = 0; i < points.length; i++) inertia += distSq(points[i], centroids[assignments[i]])

    let reassigned = assignments.length
    if (prevAssignments) {
      reassigned = 0
      for (let i = 0; i < assignments.length; i++) {
        if (assignments[i] !== prevAssignments[i]) reassigned++
      }
    }

    const centroidSnapshot = []
    for (const c of centroids) centroidSnapshot.push({x: c.x, y: c.y})
    rows.push({iter, centroids: centroidSnapshot, assignments, inertia, reassigned})

    if (prevAssignments !== null && reassigned === 0) break

    // Update step: each centroid moves to the mean of its assigned points.
    // An empty cluster (no points assigned) keeps its previous centroid
    // rather than jumping to an undefined mean.
    const sums = []
    for (let c = 0; c < k; c++) sums.push({x: 0, y: 0, count: 0})
    for (let i = 0; i < points.length; i++) {
      const c = assignments[i]
      sums[c].x += points[i].x
      sums[c].y += points[i].y
      sums[c].count++
    }
    const nextCentroids = []
    for (let c = 0; c < k; c++) {
      if (sums[c].count > 0) nextCentroids.push({x: sums[c].x / sums[c].count, y: sums[c].y / sums[c].count})
      else nextCentroids.push(centroids[c])
    }
    centroids = nextCentroids

    prevAssignments = assignments
    iter++
  }

  return {rows}
}
// runOnlineKMeans: sequential/online k-means (MacQueen's original 1967
// algorithm — the one the name "k-means" comes from). The first k points
// to arrive (in `order`) become their own singleton centroids directly;
// every later point is assigned to its nearest *current* centroid and
// immediately folded into that centroid's running mean via the
// incremental-average update μ ← μ + (x − μ)/n. Unlike Lloyd's algorithm,
// a point's assignment is never revisited once made — this single pass is
// the whole algorithm, and it's why the result depends on arrival order.
//   rows — array of { iter, pointIndex, centroids, assignments, inertia,
//   displacement }. Row 0 is the initialization state (only the first k
//   points assigned, everything else unseen — assignments hold -1 for
//   points not yet seen); each later row is the state right after one more
//   point arrived and joined a cluster.
function runOnlineKMeans(points, k, order) {
  const n = points.length
  if (n < k) return {rows: []}

  const centroids = []
  const counts = []
  const assignments = new Array(n).fill(-1)

  for (let c = 0; c < k; c++) {
    const idx = order[c]
    centroids.push({x: points[idx].x, y: points[idx].y})
    counts.push(1)
    assignments[idx] = c
  }

  const rows = [{
    iter: 0,
    pointIndex: order[k - 1],
    centroids: centroids.map(c => ({x: c.x, y: c.y})),
    assignments: assignments.slice(),
    inertia: 0,
    displacement: 0
  }]

  let cumulativeInertia = 0
  for (let step = k; step < n; step++) {
    const idx = order[step]
    const p = points[idx]

    let best = 0, bestDist = Infinity
    for (let c = 0; c < k; c++) {
      const d = distSq(p, centroids[c])
      if (d < bestDist) {
        bestDist = d
        best = c
      }
    }

    const before = {x: centroids[best].x, y: centroids[best].y}
    counts[best]++
    centroids[best] = {
      x: before.x + (p.x - before.x) / counts[best],
      y: before.y + (p.y - before.y) / counts[best]
    }
    const displacement = Math.sqrt(distSq(before, centroids[best]))

    assignments[idx] = best
    cumulativeInertia += bestDist

    rows.push({
      iter: rows.length,
      pointIndex: idx,
      centroids: centroids.map(c => ({x: c.x, y: c.y})),
      assignments: assignments.slice(),
      inertia: cumulativeInertia,
      displacement
    })
  }

  return {rows}
}
result = {
  if (points.length === 0 || k < 1) return {rows: []}
  if (mode === "Online (sequential)") return runOnlineKMeans(points, k, pointOrder)
  return runBatchKMeans(points, k, maxIterations, seedValue, regenerateTrigger)
}

}
// clusterColors: one fixed color per cluster slot (up to maxK), reused by
// both points and centroids so a centroid's "x" marker always matches the
// color of the points currently assigned to it.
clusterColors = ["#2563eb", "#dc2626", "#16a34a", "#9333ea", "#f59e0b", "#0891b2", "#db2777", "#65a30d"]
// mainPlot draws the current step's points (colored by cluster assignment)
// and centroids ("x" markers, colored to match).
//
// _plotDiv is kept in a closure so Plotly reuses the same DOM node across
// reactive updates — this preserves zoom and pan when the step changes.
mainPlot = {
  const Plotly = window.Plotly
  let _plotDiv = null

  return ({row, points, xRange, yRange}) => {
    const data = []

    if (row) {
      const byCluster = []
      for (let c = 0; c < row.centroids.length; c++) byCluster.push({xs: [], ys: []})
      const unseenXs = [], unseenYs = []
      for (let i = 0; i < points.length; i++) {
        const c = row.assignments[i]
        if (c === -1) {
          unseenXs.push(points[i].x)
          unseenYs.push(points[i].y)
        } else {
          byCluster[c].xs.push(points[i].x)
          byCluster[c].ys.push(points[i].y)
        }
      }
      if (unseenXs.length > 0) {
        data.push({
          x: unseenXs, y: unseenYs, type: "scatter", mode: "markers",
          name: "Not yet seen", showlegend: true,
          marker: {color: "#cbd5e1", size: 7, line: {color: "white", width: 0.5}}
        })
      }
      for (let c = 0; c < byCluster.length; c++) {
        data.push({
          x: byCluster[c].xs, y: byCluster[c].ys, type: "scatter", mode: "markers",
          name: `Cluster ${c + 1}`, showlegend: true,
          marker: {color: clusterColors[c % clusterColors.length], size: 8, line: {color: "white", width: 0.5}}
        })
      }

      const cxs = [], cys = [], ccolors = []
      for (let c = 0; c < row.centroids.length; c++) {
        cxs.push(row.centroids[c].x)
        cys.push(row.centroids[c].y)
        ccolors.push(clusterColors[c % clusterColors.length])
      }
      data.push({
        x: cxs, y: cys, type: "scatter", mode: "markers", name: "Centroids", showlegend: true,
        marker: {color: ccolors, size: 16, symbol: "x", line: {color: "#111827", width: 2}}
      })
    }

    const layout = {
      margin: {l: 0, r: 0, t: 0, b: 0},
      xaxis: {title: "x", range: [xRange.lo, xRange.hi], zeroline: true},
      // scaleanchor/scaleratio lock the y-axis to the same scale as x, so
      // clusters that are geometrically round don't render as skewed
      // ellipses just because the plot box isn't square.
      yaxis: {title: "y", range: [yRange.lo, yRange.hi], zeroline: true, scaleanchor: "x", scaleratio: 1},
      legend: {orientation: "h", y: -0.08},
      hovermode: "closest",
      uirevision: "static",
      autosize: true
    }
    const config = {responsive: true, displaylogo: false}

    if (!_plotDiv) {
      _plotDiv = document.createElement("div")
      _plotDiv.className = "plotly-box-large"
      Plotly.newPlot(_plotDiv, data, layout, config)
      // newPlot fires before the div is in the DOM, so Plotly measures 0 width.
      // A ResizeObserver catches the real size once it's inserted and laid out,
      // and keeps correcting it on any later layout/viewport change.
      new ResizeObserver(() => Plotly.Plots.resize(_plotDiv)).observe(_plotDiv)
    } else {
      Plotly.react(_plotDiv, data, layout, config)
    }
    return _plotDiv
  }
}
mainPlot(setup)

// viewof mode: Batch (Lloyd's algorithm — the version implemented earlier)
// alternates a full assignment pass and a full centroid-update pass over
// every point, repeated to convergence. Online (sequential/MacQueen's
// algorithm, the version the name "k-means" originally comes from) instead
// feeds points in one at a time, updating one centroid's running mean
// immediately after each point arrives — see the formula section below.
// This only changes how the current fixed set of points is clustered, not
// the points themselves, so — unlike trueGroups/seedValue above — it lives
// here on the chart rather than in the panel.

  // format shortens the two options' displayed text to fit the overlay's

  // shown, not mode's actual value, so every comparison against the full
  // "Batch (Lloyd's)"/"Online (sequential)" strings elsewhere on this page
  // (result, urlSyncControls, examples) is unaffected.
  const shortLabel = (d) => d === "Online (sequential)" ? "Online" : "Batch"

  input.dataset.exampleField = "mode"

// iteratesPlot: inertia (the k-means cost function) vs step, for the
// entire run -- doesn't depend on the current step, so it's plotted once
// per run. Lloyd's algorithm (Batch) computes this over every point each
// iteration and never lets it increase; Online k-means only ever has
// inertia accumulated over the points seen SO FAR, which is a genuinely
// different quantity (not just a differently-scoped one), so the y-axis
// label says which is shown.
iteratesPlot = {
  const isOnline = mode === "Online (sequential)"
  return Plot.plot({

K-means partitions \(n\) points into \(k\) clusters by minimizing the total squared distance from each point to its cluster’s centroid — the inertia,

\[ J = \sum_{i=1}^n \big\lVert x_i - \mu_{z_i} \big\rVert^2, \]

where \(z_i \in \{1, \dots, k\}\) is the cluster point \(x_i\) is assigned to and \(\mu_c\) is cluster \(c\)’s centroid. Lloyd’s algorithm finds a local minimum of \(J\) by alternating two steps until nothing changes:

  1. Assignment step — fix the centroids, and assign every point to its nearest one: \(z_i = \operatorname*{arg\,min}_c \lVert x_i - \mu_c \rVert^2\).
  2. Update step — fix the assignments, and move each centroid to the mean of the points now assigned to it: \(\mu_c = \frac{1}{|\{i : z_i = c\}|}\sum_{i : z_i = c} x_i\).

Neither step can ever increase \(J\), so the algorithm always converges — but only to a local minimum, and which one depends on where the centroids started.

Online (sequential) k-means — MacQueen’s original 1967 algorithm, the one the name “k-means” actually comes from — never makes a full pass over all the data at once. The first \(k\) points to arrive become the initial centroids directly; every point after that is assigned to its nearest current centroid and immediately folded into that one centroid’s running mean, with \(n_c\) the number of points assigned to cluster \(c\) so far (including this one):

\[ \mu_c \leftarrow \mu_c + \frac{1}{n_c}\big(x - \mu_c\big). \]

A point’s assignment, once made, is never revisited — there is no separate assignment/update pass, just one point at a time, once each. That’s what makes the result depend on the order points arrive in, unlike Lloyd’s algorithm.
